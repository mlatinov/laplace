    # How `laplace sampler test` runs langevin::barker (laplace-sample suite.rs).
    #
    # Barker passes simulation-based calibration (200 replicates, uniform ranks
    # on all ten quantities), so a target that does not converge needs more
    # draws, never a looser threshold. Its moves are bounded and per-coordinate,
    # and under adapt() its acceptance settles at about 0.25-0.37, below the
    # 0.4 target: dual averaging assumes acceptance falls steadily with the step
    # size, and Barker's barely moves. So it mixes slowly and needs long runs.
    kernel = "barker"
    adapt = true

    warmup = 1000
    draws = 10000
    thin = 10
